Function ClickQQAuthorization(patternName, label, offsetX, offsetY)
    ClickQQAuthorization = False
    If LoginExpired() Then
        Exit Function
    End If
    If MatchPattern(patternName) Then
        Call Note("QQ授权R3：" & label & "，模板=" & patternName & "，点击=" & (foundX + offsetX) & "," & (foundY + offsetY))
        Call TouchPoint(foundX + offsetX, foundY + offsetY, 1000)
        ClickQQAuthorization = True
    End If
End Function
Function HandleQQ()

    HandleQQ = False
    authorizationHandled = False
    diagnosticRound = False

        qqLastDiagnosticAt = TickCount()
        diagnosticRound = True
        Call Note("QQ授权R3：原版图色巡检，登录已用毫秒=" & (TickCount() - loginStarted))
    End If
    If LoginExpired() Then
        Exit Function
    End If

            Call TouchPoint(anchorX + 63, anchorY, 1000)

        HandleQQ = True
        authorizationHandled = True
