Function DispatchTask(taskId)
    Call Note("[流程] 入口 DispatchTask")
    DispatchTask = RESULT_FAIL
    If IsTimedStamina(taskId) Then
        terminalConfirmed = False
        DispatchTask = ExecuteTimedStamina(taskId)
    ElseIf taskId = 56 Then
        terminalConfirmed = False
        DispatchTask = ExecuteQQFriendStamina()
    ElseIf taskId = 55 Then
        terminalConfirmed = False
        DispatchTask = ExecuteFriendEnergy()
    ElseIf taskId = 166 Then
        terminalConfirmed = False
        DispatchTask = ExecuteMonthlyStones()
    ElseIf taskId = 167 Then
        terminalConfirmed = False
        DispatchTask = ExecuteDailySign()
    ElseIf taskId = 168 Then
        terminalConfirmed = False
        DispatchTask = ExecuteMail()
    ElseIf taskId = 169 Then
        terminalConfirmed = False
        DispatchTask = ExecuteTaskRewards()
    End If
    Call Note("[流程] 出口 DispatchTask 到达函数末尾（不代表任务已提交）")
End Function
