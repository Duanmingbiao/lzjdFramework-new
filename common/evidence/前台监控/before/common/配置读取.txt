Function PositiveSetting(key, fallback)
    Dim raw
    raw = ReadUIConfig(key, fallback)
    PositiveSetting = fallback
    If IsNumeric(raw) Then
        If CDbl(raw) > 0 Then
            PositiveSetting = CDbl(raw)
        End If
    End If
End Function
Sub LoadSettings()
    rewardEnabled = ReadUIConfig("任务奖励启用", True)
    rewardInterval = PositiveSetting("任务奖励间隔秒", 21600)
    mailEnabled = ReadUIConfig("邮件启用", True)
    mailInterval = PositiveSetting("邮件间隔秒", 21600)
    loginLimit = PositiveSetting("登录总超时秒", 600) * 1000
    businessLimit = PositiveSetting("业务总超时秒", 120) * 1000
    messageMode = ReadUIConfig("消息邮件", 0)
    // root.784: zero means 1000ms; positive values are seconds.
    hallReturnDelaySeconds = PositiveSetting("返回延迟秒", 0)
End Sub
